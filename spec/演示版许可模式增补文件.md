1. 增补说明
1.1 增补目的
为支持产品演示、用户培训和商务推广，系统需支持演示版许可模式（Demo License）。演示版通过模块级记录数限制、项目数量限制、CONFIG只读和水印机制，在提供完整功能体验的同时防止被用作实际生产工具。

试用版	TRIAL	客户限时试用（可选扩展）	30天有效期，到期转DEMO

3.3.16.10 计数口径（V1.1 新增，ADR-0011）

1. 活记录定义：sign_status ≠ OBSOLETE 的所有记录。计数范围：DRAFT / IN_APPROVAL / CHECKED / CHECK_REJECTED / STALE / CHANGE_PENDING / CHANGED / REVERSAL_PENDING；物流按 StreamSignStatus 同理（非 OBSOLETE 计数）。

2. 弃用返还：记录弃用（→OBSOLETE）后配额自动返还（计数查询排除 OBSOLETE，天然返还），演示用户可在配额内"创建→弃用→再创建"循环。

3. 不计数对象：交付物（含变更单）、交付物版本；报表定义独立限制（≤3 个，维持本表）。

4. 设备表：无独立限制，跟随来源模块的活记录计数（EQUIP_LIST 不在 module_limits 中）。

5. 配额提示展示"X/3 条活记录"，达到限制时引导弃用腾配额。

# app/core/license_decorator.py（P1阶段完成）
from functools import wraps
from fastapi import HTTPException
from app.core.license import get_license_config, LicenseType
from app.services.license_counter import count_records_in_module

def check_license_limit(module: str):
    """
    通用许可限制装饰器
    使用方式：在路由处理函数上添加 @check_license_limit("PUMP")
    """
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            license_config = get_license_config()
            
            # 完整版不检查
            if license_config.license_type == LicenseType.FULL:
                return await func(*args, **kwargs)
            
            # 演示版/试用版检查计数
            project_id = kwargs.get('project_id') or _extract_project_id(args)
            current_count = await count_records_in_module(module, project_id)
            max_count = license_config.module_limits.get(module, 3)
            
            if current_count >= max_count:
                # 记录审计日志
                await log_license_violation(
                    user_id=_get_current_user_id(),
                    module=module,
                    action="CREATE",
                    reason=f"LIMIT_EXCEEDED: {current_count}/{max_count}"
                )
                raise HTTPException(
                    status_code=403,
                    detail=f"演示版限制：{module}模块最多允许{max_count}条记录"
                )
            
            return await func(*args, **kwargs)
        return wrapper
    return decorator

class LicenseCheckMixin:
    """所有业务服务基类，统一许可检查"""
    
    @property
    def license_config(self):
        return get_license_config()
    
    def check_module_limit(self, module: str, current_count: int):
        """模块记录数限制检查"""
        if self.license_config.license_type != LicenseType.FULL:
            max_count = self.license_config.module_limits.get(module, 3)
            if current_count >= max_count:
                raise DemoLimitExceeded(
                    module=module,
                    current=current_count,
                    max=max_count
                )
    
    def check_config_writable(self):
        """CONFIG只读检查"""
        if self.license_config.config_readonly:
            raise ConfigReadOnlyError()
    
    def check_equip_lib_settle_allowed(self):
        """EQUIP_LIB沉淀检查"""
        if self.license_config.equip_lib_settle_disabled:

CREATE OR REPLACE FUNCTION check_demo_pump_limit()
RETURNS TRIGGER AS $$
DECLARE
    current_count INTEGER;
    max_count INTEGER;
BEGIN
    SELECT (module_limits_json->>'PUMP')::INTEGER INTO max_count
    FROM license_configs
    WHERE license_type = 'DEMO'
    LIMIT 1;
    
    IF max_count IS NOT NULL THEN
        SELECT COUNT(*) INTO current_count
        FROM pump_results
        WHERE project_id = NEW.project_id;
        
        IF current_count >= max_count THEN
            RAISE EXCEPTION 'DEMO_LIMIT_EXCEEDED: PUMP max %', max_count;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER demo_limit_pump_insert
BEFORE INSERT ON pump_results
FOR EACH ROW
EXECUTE FUNCTION check_demo_pump_limit();
