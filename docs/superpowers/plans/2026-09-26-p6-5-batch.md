```python
# app/services/pipe/two_phase_erosion.py
"""API 14E 两相流冲蚀速度（SPEC §3.2.1 V1.8）。

公式：V_e = C / √ρ_mix（API 14E 5th Ed §4.3.2 Eq.4）
   C 因子（按服务类型，SPEC §3.2.1 表 3.2.1）：
     连续 service = 122
     间歇 service = 152.5
     耐蚀连续     = 183~244（默认 200）
     耐蚀间歇     = 305
   ρ_mix：两相流体积加权平均密度

Imperial 双单位（仅 imperial_units=True）：
   V_e(ft/s) = V_e(m/s) / 0.3048

Q-5（OpenWolf 登记）：C 因子 CONFIG 表未建，本批用 SYNTHETIC_TEST_DATA + confirmed_by 占位
"""
from __future__ import annotations
import math
from dataclasses import dataclass
from typing import Final, Literal
from app.services.exceptions import PcsError

# C 因子默认值（按服务类型，SPEC §3.2.1 表 3.2.1 SI）
ServiceType = Literal["CONTINUOUS", "INTERMITTENT", "CORROSIVE_CONTINUOUS", "CORROSIVE_INTERMITTENT"]
_C_FACTOR_BY_SERVICE: Final[dict[str, float]] = {
    "CONTINUOUS": 122.0,
    "INTERMITTENT": 152.5,
    "CORROSIVE_CONTINUOUS": 200.0,
    "CORROSIVE_INTERMITTENT": 305.0,
}

class Api14eErosionInputError(PcsError):
    code = "API14E_EROSION_INPUT_ERROR"
    status = 422

@dataclass(frozen=True)
class Api14eErosionInput:
    rho_mix_kg_m3: float
    service_type: ServiceType  # 决定 C 因子默认；显式 c_factor 非 None 时覆盖
    c_factor: float | None = None  # None 时按 service_type 查 _C_FACTOR_BY_SERVICE
    mass_flow_kg_s: float = 0.0
    pipe_diameter_m: float = 0.1
    imperial_units: bool = False

@dataclass(frozen=True)
class Api14eErosionResult:
    v_e_m_s: float
    actual_v_m_s: float
    is_erosion_safe: bool
    imperial_conversion: dict[str, float] | None
    formula_ref: dict[str, str]

def _resolve_c_factor(inp: Api14eErosionInput) -> float:
    """解析 C 因子：显式 c_factor 非 None 时覆盖，否则按 service_type 查 dict。"""
    if inp.c_factor is not None:
        return inp.c_factor
    return _C_FACTOR_BY_SERVICE[inp.service_type]

def _validate_input(inp: Api14eErosionInput) -> None:
    if inp.rho_mix_kg_m3 <= 0:
        raise Api14eErosionInputError(f"ρ_mix={inp.rho_mix_kg_m3} 必须 > 0")
    c = _resolve_c_factor(inp)
    if not (50.0 <= c <= 400.0):
        raise Api14eErosionInputError(f"C 因子={c} 越界（API 14E 范围 50~400）")
    if inp.mass_flow_kg_s < 0:
        raise Api14eErosionInputError(f"mass_flow={inp.mass_flow_kg_s} 不能为负")
    if inp.pipe_diameter_m <= 0:
        raise Api14eErosionInputError(f"diameter={inp.pipe_diameter_m} 必须 > 0")

def calc_api14e_erosion_velocity(inp: Api14eErosionInput) -> Api14eErosionResult:
    _validate_input(inp)
    c = _resolve_c_factor(inp)
    v_e = c / math.sqrt(inp.rho_mix_kg_m3)
    q_m3_s = inp.mass_flow_kg_s / inp.rho_mix_kg_m3
    a_m2 = math.pi * inp.pipe_diameter_m**2 / 4.0
    v_actual = q_m3_s / a_m2
    is_safe = v_actual < v_e

    imperial = None
    if inp.imperial_units:
        imperial = {
            "v_e_ft_s": v_e / 0.3048,
            "actual_v_ft_s": v_actual / 0.3048,
        }

    return Api14eErosionResult(
        v_e_m_s=v_e,
        actual_v_m_s=v_actual,
        is_erosion_safe=is_safe,
        imperial_conversion=imperial,
        formula_ref={
            "erosion_velocity": "V_e = C / √ρ_mix [API 14E 5th Ed §4.3.2 Eq.4]",
            "c_factor_by_service": "SPEC §3.2.1 表 3.2.1: CONTINUOUS=122 / INTERMITTENT=152.5 / CORROSIVE_CONTINUOUS=183~244 / CORROSIVE_INTERMITTENT=305",
        },
    )

__all__ = [
    "Api14eErosionInput",
    "Api14eErosionResult",
    "Api14eErosionInputError",
    "calc_api14e_erosion_velocity",
]
```

```python
"""API 14E 两相流管道尺寸（SPEC §3.2.2 V1.1）。

公式：
  ρ_mix = ρ_L·(1-ε) + ρ_V·ε
  V_e = C / √ρ_mix（API 14E Eq.4）
  D_min = √(4·m / (π·ρ_mix·V_e))（连续性方程）

校核：
  - 临界携液（Turner 1966）：V_g ≥ V_turner = 5.46·(σ·g·(ρ_L-ρ_V)/ρ_V²)^0.25
  - 段塞捕集（按 C-15 Beggs-Brill 持液率）

Q-6：临界携液 σ 默认 0.02 N/m（水-气），CONFIG 表 P6-14 接管真实界面张力
"""
from __future__ import annotations
import math
from dataclasses import dataclass
from typing import Final
from app.services.exceptions import PcsError

_SURFACE_TENSION_WATER_AIR_DEFAULT: Final[float] = 0.02  # N/m（水-气 25°C）
_GRAVITY: Final[float] = 9.81

class Api14ePipeSizingInputError(PcsError):
    code = "API14E_SIZING_INPUT_ERROR"
    status = 422

@dataclass(frozen=True)
class Api14ePipeSizingInput:
    mass_flow_kg_s: float
    rho_L_kg_m3: float
    rho_V_kg_m3: float
    void_fraction: float
    c_factor: float
    surface_tension_n_m: float = _SURFACE_TENSION_WATER_AIR_DEFAULT
    imperial_units: bool = False

@dataclass(frozen=True)
class Api14ePipeSizingResult:
    d_min_m: float
    v_e_m_s: float
    v_turner_m_s: float  # 临界携液速度（Turner 1966）
    is_liquid_unloading_safe: bool
    imperial_conversion: dict[str, float] | None
    formula_ref: dict[str, str]

def _validate_input(inp: Api14ePipeSizingInput) -> None:
    if inp.mass_flow_kg_s <= 0:
        raise Api14ePipeSizingInputError(f"m={inp.mass_flow_kg_s} 必须 > 0")
    if inp.rho_L_kg_m3 <= 0 or inp.rho_V_kg_m3 <= 0:
        raise Api14ePipeSizingInputError("密度必须 > 0")
    if not (0.0 <= inp.void_fraction <= 1.0):
        raise Api14ePipeSizingInputError(f"void_fraction={inp.void_fraction} 越界 [0,1]")
    if not (50.0 <= inp.c_factor <= 250.0):
        raise Api14ePipeSizingInputError(f"C 因子={inp.c_factor} 越界")

def calc_api14e_pipe_size(inp: Api14ePipeSizingInput) -> Api14ePipeSizingResult:
    _validate_input(inp)
    rho_mix = inp.rho_L_kg_m3 * (1 - inp.void_fraction) + inp.rho_V_kg_m3 * inp.void_fraction
    v_e = inp.c_factor / math.sqrt(rho_mix)
    # D_min = √(4m / (πρV))
    d_min = math.sqrt(4.0 * inp.mass_flow_kg_s / (math.pi * rho_mix * v_e))
    # Turner 1966 临界携液速度（仅气相主导；液相主导返回 0 + safe）
    if inp.void_fraction > 0.5:
        v_turner = 5.46 * (
            inp.surface_tension_n_m * _GRAVITY * (inp.rho_L_kg_m3 - inp.rho_V_kg_m3)
            / inp.rho_V_kg_m3**2
        ) ** 0.25
        v_g = inp.mass_flow_kg_s / (inp.rho_V_kg_m3 * math.pi * d_min**2 / 4.0)
        is_safe = v_g >= v_turner
    else:
        v_turner = 0.0
        is_safe = True

    imperial = None
    if inp.imperial_units:
        imperial = {
            "d_min_in": d_min / 0.0254,
            "v_e_ft_s": v_e / 0.3048,
            "v_turner_ft_s": v_turner / 0.3048,
        }

    return Api14ePipeSizingResult(
        d_min_m=d_min,
        v_e_m_s=v_e,
        v_turner_m_s=v_turner,
        is_liquid_unloading_safe=is_safe,
        imperial_conversion=imperial,
        formula_ref={
            "rho_mix": "ρ_mix = ρ_L(1-ε) + ρ_V·ε",
            "erosion_velocity": "V_e = C/√ρ_mix [API 14E Eq.4]",
            "pipe_diameter": "D_min = √(4m/(πρV)) [连续性]",
            "turner_critical_velocity": "V_turner = 5.46·(σg(ρ_L-ρ_V)/ρ_V²)^0.25 [Turner 1966]",
        },
    )
```

```python
    a = a_f / √(1 + (K·D)/(E·e)·C₁)
    E：管材弹性模量（碳钢 200 GPa）
    e：壁厚
    C₁：管锚固系数（1 - ν²；ν=0.3 → C₁=0.91；两端锚固）
  瞬时关阀（Joukowsky 1898）：ΔP = ρ·a·ΔV
  缓慢关阀（MOC/特征线法）：t_c ≥ 2L/a 时无水锤
  段塞捕集（按 C-15 Beggs-Brill 持液率 → 段塞体积 V_s = L·A·(1-H_L(θ))

Q-7：段塞捕集器尺寸走 C-15 Beggs-Brill 接口（Task A4）

_PIPE_E_MODULUS_PA: Final[dict] = {
    "CARBON_STEEL": 200e9,
    "STAINLESS": 195e9,
    "DUCTILE_IRON": 165e9,
    "PVC": 3e9,

    if inp.fluid_density_kg_m3 <= 0:
        raise SurgePressureInputError(f"ρ={inp.fluid_density_kg_m3} 必须 > 0")
    if inp.fluid_bulk_modulus_pa <= 0:
        raise SurgePressureInputError(f"K={inp.fluid_bulk_modulus_pa} 必须 > 0")
    if inp.pipe_diameter_m <= 0:
        raise SurgePressureInputError(f"D={inp.pipe_diameter_m} 必须 > 0")
    if inp.flow_velocity_m_s < 0:
        raise SurgePressureInputError(f"V={inp.flow_velocity_m_s} 不能为负")
    if inp.valve_close_time_s < 0:
        raise SurgePressureInputError(f"t_c={inp.valve_close_time_s} 不能为负")
    if inp.wall_thickness_m < 0:
        raise SurgePressureInputError(f"e={inp.wall_thickness_m} 不能为负")
    if inp.pipe_material not in _PIPE_E_MODULUS_PA:
        raise SurgePressureInputError(
            f"材质 {inp.pipe_material} 必须是 {_PIPE_E_MODULUS_PA.keys()} 之一"
        )

def _wave_speed_with_wall_correction(inp: SurgePressureInput) -> tuple[float, float]:
    """返回 (a, a_fluid)：a 含管壁修正；a_fluid 纯流体声速。

    Wylie & Streeter 1993：a = a_f / √(1 + (K·D)/(E·e)·C₁)
    e=0 时 → 纯流体声速（向后兼容）。
    """
    a_f = math.sqrt(inp.fluid_bulk_modulus_pa / inp.fluid_density_kg_m3)
    if inp.wall_thickness_m <= 0:
        return a_f, a_f
    E = _PIPE_E_MODULUS_PA[inp.pipe_material]
    # 修正项：(K·D)/(E·e)·C₁
    correction = (
        inp.fluid_bulk_modulus_pa * inp.pipe_diameter_m
        / (E * inp.wall_thickness_m) * _C1_ANCHOR_COEFF
    )
    a = a_f / math.sqrt(1.0 + correction)
    return a, a_f

def calc_water_hammer_surge(inp: SurgePressureInput) -> SurgePressureResult:
    _validate_input(inp)
    a, a_fluid = _wave_speed_with_wall_correction(inp)
    # Joukowsky：瞬时关阀（t_c → 0）；用含壁修正的 a
    delta_p = inp.fluid_density_kg_m3 * a * inp.flow_velocity_m_s
    # 临界关阀时间 t_c_critical = 2L/a
    t_c_critical = 2.0 * inp.pipe_length_m / a
    is_joukowsky = inp.valve_close_time_s < t_c_critical

    imperial = None
    if inp.imperial_units:

            "wave_speed_ft_s": a / 0.3048,
            "surge_pressure_psi": delta_p / 6894.76,
            "critical_close_time_s": t_c_critical,
        }

    return SurgePressureResult(
        wave_speed_m_s=a,
        surge_pressure_pa=delta_p,
        critical_close_time_s=t_c_critical,
        is_joukowsky_applicable=is_joukowsky,
        imperial_conversion=imperial,
        formula_ref={
            "wave_speed": "a = √(K/ρ) [水锤声速]",
            "joukowsky": "ΔP = ρ·a·ΔV [Joukowsky 1898 瞬时关阀]",
            "critical_close_time": "t_c_critical = 2L/a [缓慢关阀边界]",
        },
    )

    "SurgePressureInput",
    "SurgePressureResult",

]
```

```python
算法清单：
  1. 流型判别：Mandhane 1975 水平管流型图（BUBBLE/STRATIFIED/STRATIFIED_WAVE/ANNULAR/INTERMITTENT/DISPERSED BUBBLE；BG-B 1973 原文引用）

  3. 流型校验：Eaton-Flanning 1967（对比 BG-B 适用范围）

  Fr = V_m² / (g·D₁)  弗劳德数
  L₁ = 316·ρ_L^0.302  L₂ = 0.000925·(ρ_L·ρ_V)^0.305  L₃ = 0.1·ρ_L^-0.447
  流型判别（水平管）：
    - 分离流：λ_L < L₁ 且 λ_G < L₃
    - 间歇流：λ_L ≥ L₂ 且 λ_G ≥ L₃
    - 分散流：λ_G ≥ L₁ 且 λ_G < L₃ 且 λ_L ≥ L₃
  H_L(0) = a·λ_L^b / Fr^c  （流型参数 a/b/c 查表）

  H_L(θ) = H_L(0)·C(θ)

    """Mandhane 1975 水平管流型图（SPEC §3.2.3 V1.9）。

    注：L1/L2/L3 阈值源自 Mandhane, G.A., et al. (1975) "A Flow Pattern Map
    for Horizontal Two-Phase Flow"，被 Beggs-Brill 1973 引用为适用范围校验。
    非 Taitel-Dukler 1976（后者用 K/T/F/X 4 个无量纲数）。
    """
    L1 = 316.0 * rho_L**0.302
    L2 = 0.000925 * (rho_L * rho_V) ** 0.305
    L3 = 0.1 * rho_L ** (-0.447)
    if lambda_V < L3 and lambda_L < L1:
        return "SEGREGATED"
    elif lambda_L >= L2 and lambda_V >= L3:
        return "INTERMITTENT"
    elif lambda_V >= L1 and lambda_V < L3 and lambda_L >= L3:
        return "DISTRIBUTED"
    else:
        return "TRANSITION"

def _beggs_brill_h_l_theta0(
    lambda_L: float, fr: float, flow_pattern: FlowPattern,
) -> float:
    """Beggs-Brill 1973 Eq.6 H_L(0) = a·λ_L^b / Fr^c。"""
    # 系数表（Beggs-Brill 1973 Table 1）
    COEFFS = {
        "SEGREGATED": (0.980, 0.4846, 0.0868),
        "INTERMITTENT": (0.845, 0.5351, 0.0173),
        "DISTRIBUTED": (1.065, 0.5824, 0.0609),
    }
    if flow_pattern == "TRANSITION":
        # 边界流型取 SEGREGATED/INTERMITTENT 平均
        a, b, c = (0.913, 0.5100, 0.0521)
    else:
        a, b, c = COEFFS[flow_pattern]
    if fr <= 0:
        return lambda_L
    return a * lambda_L**b / fr**c

def _beggs_brill_inclination_correction(
    h_l_theta0: float, theta_deg: float, fr: float, v_m: float,
) -> float:
    """倾角修正 B(θ) = 1 + (C-1)·(sin θ)^d（BG-B 1973 Eq.14-16 简化）。"""
    if abs(theta_deg) < 1e-6:
        return h_l_theta0
    theta_rad = math.radians(theta_deg)
    # 上坡（θ>0）：H_L 增加；下坡（θ<0）：H_L 减少
    sign = 1.0 if theta_deg > 0 else -0.5
    correction = 1.0 + sign * 0.5 * math.sin(theta_rad) ** 1.5
    return max(0.0, min(1.0, h_l_theta0 * correction))

def calc_beggs_brill_holdup(inp: BeggsBrillHoldupInput) -> HoldupCorrelationResult:
    _validate_input(inp)
    v_m = inp.v_sl_m_s + inp.v_sg_m_s
    a_pipe = math.pi * inp.pipe_diameter_m**2 / 4.0
    q_L = inp.v_sl_m_s * a_pipe
    q_V = inp.v_sg_m_s * a_pipe
    q_m = q_L + q_V
    lambda_L = q_L / q_m if q_m > 0 else 0.0
    lambda_V = q_V / q_m if q_m > 0 else 1.0
    fr = v_m**2 / (_GRAVITY * inp.pipe_diameter_m) if inp.pipe_diameter_m > 0 else 0.0

    flow_pattern = _mandhane_flow_pattern(lambda_L, lambda_V, inp.rho_L_kg_m3, inp.rho_V_kg_m3)
    h_l_0 = _beggs_brill_h_l_theta0(lambda_L, fr, flow_pattern)
    h_l_theta = _beggs_brill_inclination_correction(h_l_0, inp.pipe_inclination_deg, fr, v_m)

    # Eaton-Flanning 1967 校验：E-F 适用范围 Fr ∈ [0.01, 10]
    is_valid = 0.01 <= fr <= 10.0

    imperial = None
    if inp.imperial_units:
        imperial = {
            "v_sl_ft_s": inp.v_sl_m_s / 0.3048,
            "v_sg_ft_s": inp.v_sg_m_s / 0.3048,
        }

    return HoldupCorrelationResult(
        flow_pattern=flow_pattern,
        h_l_theta0=h_l_0,
        h_l_theta=h_l_theta,
        froude_number=fr,
        is_beggs_brill_valid=is_valid,
        imperial_conversion=imperial,
        formula_ref={
            "mandhane_flow_pattern": "Mandhane 1975 水平管流型图（SPEC §3.2.3 V1.9：原 Taitel-Dukler → Mandhane 1975，因 BG-B 1973 原文引用 Mandhane 而非 Taitel-Dukler；Taitel-Dukler 1976 用 K/T/F/X 4 无量纲数不同算法，本批不实现）",

            "eaton_flanning": "Eaton-Flanning 1967 Fr 范围校验 [0.01, 10]",
        },
    )
```

```python
  声速 a = √(γ·R·T/M) = √(γ·P/ρ)（理想气体）

from dataclasses import dataclass

from app.services.exceptions import PcsError

class As2360InputError(PcsError):

    status = 422

@dataclass(frozen=True)
class As2360LimitInput:
    orifice_diameter_m: float
    inlet_pressure_kpa: float
    outlet_pressure_kpa: float

    orifice_diameter_in: float
    is_limit_applicable: bool
    sonic_velocity_m_s: float
    critical_pressure_ratio: float
    actual_pressure_ratio: float
    is_choked: bool
    warning_message: str | None
    imperial_conversion: dict[str, float] | None
    formula_ref: dict[str, str]

def _validate_input(inp: As2360LimitInput) -> None:
    if inp.orifice_diameter_m <= 0:
        raise As2360InputError(f"d={inp.orifice_diameter_m} 必须 > 0")
    if inp.inlet_pressure_kpa <= 0:
        raise As2360InputError(f"P₁={inp.inlet_pressure_kpa} 必须 > 0")
    if inp.outlet_pressure_kpa < 0:
        raise As2360InputError(f"P₂={inp.outlet_pressure_kpa} 不能为负")
    if inp.fluid_density_kg_m3 <= 0:
        raise As2360InputError("密度必须 > 0")
    if not (1.05 <= inp.gas_specific_heat_ratio <= 1.67):
        raise As2360InputError(f"γ={inp.gas_specific_heat_ratio} 越界 [1.05, 1.67]")

def calc_as_2360_1_1_limit(inp: As2360LimitInput) -> As2360LimitResult:
    _validate_input(inp)
    d_in = inp.orifice_diameter_m / 0.0254
    is_limit = d_in < 1.0
    # 声速（理想气体）：a = √(γ·P₁/ρ)（P₁ 为绝对压力）
    p1_pa = inp.inlet_pressure_kpa * 1000.0
    a = math.sqrt(inp.gas_specific_heat_ratio * p1_pa / inp.fluid_density_kg_m3)
    # 临界压比 r_c = (2/(γ+1))^(γ/(γ-1))
    gamma = inp.gas_specific_heat_ratio
    r_c = (2.0 / (gamma + 1.0)) ** (gamma / (gamma - 1.0))
    # 实际压比 P₂/P₁
    p_ratio = inp.outlet_pressure_kpa / inp.inlet_pressure_kpa
    # 阻塞判断：P₂/P₁ ≤ r_c
    is_choked = p_ratio <= r_c

    warning = None
    if is_limit:
        warning = (
            f"AS 2360.1.1 Limit: orifice d={d_in:.3f}in < 1in，"
            f"建议查表 AS 2360.1.1 Limit Factor C_L（CONFIG 表未命中，"
            f"使用合成占位 Q-9）"
        )

    imperial = None
    if inp.imperial_units:
        imperial = {
            "sonic_velocity_ft_s": a / 0.3048,
            "inlet_pressure_psia": inp.inlet_pressure_kpa * 0.1450377,
        }

    return As2360LimitResult(
        orifice_diameter_in=d_in,
        is_limit_applicable=is_limit,
        sonic_velocity_m_s=a,
        critical_pressure_ratio=r_c,
        actual_pressure_ratio=p_ratio,
        is_choked=is_choked,
        warning_message=warning,
        imperial_conversion=imperial,
        formula_ref={
            "sonic_velocity": "a = √(γ·P₁/ρ) [理想气体声速]",
            "critical_pressure_ratio": "r_c = (2/(γ+1))^(γ/(γ-1)) [等熵临界压比]",

            "as_standard": "AS 2360.1.1-1993 (R2016) §4.2.3 Limit Factor",
        },
    )
```

```python
def _validate_input(inp: ThreePhaseSeparatorInput) -> None:
    if inp.diameter_m <= 0 or inp.length_m < 0:
        raise ThreePhaseSeparatorError("几何尺寸必须 D>0, L≥0")
    for fld in ("oil_mass_rate_kg_d", "water_mass_rate_kg_d", "gas_mass_rate_kg_d"):
        if getattr(inp, fld) < 0:
            raise ThreePhaseSeparatorError(f"{fld} 不能为负")
    if inp.oil_density_kg_m3 <= 0 or inp.water_density_kg_m3 <= 0 or inp.gas_density_kg_m3 <= 0:

    if not (0.0 < inp.weir_height_m < inp.diameter_m):
        raise ThreePhaseSeparatorError(f"weir_height={inp.weir_height_m} 越界 (0, D)")

def calc_three_phase_separator(inp: ThreePhaseSeparatorInput) -> ThreePhaseSeparatorResult:
    _validate_input(inp)
    # 段 1：体积流量
    q_oil = inp.oil_mass_rate_kg_d / inp.oil_density_kg_m3
    q_water = inp.water_mass_rate_kg_d / inp.water_density_kg_m3
    q_gas = inp.gas_mass_rate_kg_d / inp.gas_density_kg_m3
    q_liquid = q_oil + q_water  # m³/d

    # 段 3：油水界面（SPEC §3.4.3 Eq.1）
    # H_ow = H_total·(ρ_w - ρ_oil)/(ρ_w - ρ_g)
    # 反推 H_total：让 h_ow == target（设计意图：H_target 是用户给的界面高度）
    # H_total = H_target · (ρ_w - ρ_g)/(ρ_w - ρ_oil)
    h_total = inp.oil_water_interface_target_m * (
        inp.water_density_kg_m3 - inp.gas_density_kg_m3
    ) / (inp.water_density_kg_m3 - inp.oil_density_kg_m3)
    # 几何约束（按 vessel_shape 区分上限）：
    #   HORIZONTAL: H_total ≤ D（液位受 D 限制）
    #   VERTICAL: H_total ≤ D + 2·head_depth（液位可进入顶部封头）
    # 封头深度近似：HEMISPHERICAL = D/2；2:1_ELLIPTICAL = D/4；TORISPHERICAL ≈ 0.169·D
    # Z-2 v5：Final 仅用于模块级/类级；函数体内局部变量移除 Final（mypy 告警）
    _HEAD_DEPTH_M = {
        "HEMISPHERICAL": lambda D: D / 2.0,
        "2:1_ELLIPTICAL": lambda D: D / 4.0,
        "TORISPHERICAL": lambda D: 0.169 * D,
        "FLAT": lambda D: 0.0,
    }
    head_depth = _HEAD_DEPTH_M[inp.head_type](inp.diameter_m)
    if inp.vessel_shape == "HORIZONTAL":
        h_max = inp.diameter_m
    else:  # VERTICAL
        h_max = inp.diameter_m + 2.0 * head_depth
    if h_total > h_max:
        raise ThreePhaseSeparatorError(
            f"H_total={h_total:.3f}m > h_max={h_max:.3f}m "
            f"({inp.vessel_shape} 上限：D{'+2·head_depth' if inp.vessel_shape == 'VERTICAL' else ''}) "
            f"目标界面 {inp.oil_water_interface_target_m}m 超出几何"
        )
    h_ow = inp.oil_water_interface_target_m  # 由反推保证 h_ow == target
    # 堰板高度 = h_ow + 安全裕度（典型 0.1-0.3 m，本计划用 inp.weir_height_m 作为设计值）
    h_w_required = inp.weir_height_m

    # 段 4-5：停留时间（用 C-12 calc_partial_volume 算 total）
    pv_inp = PartialVolumeInput(
        D_m=inp.diameter_m, L_m=inp.length_m, head_type=inp.head_type,
        H_m=h_total, n_vessels=1,
    )
    pv_result = calc_partial_volume(pv_inp)
    v_total = pv_result.total_volume_m3
    # 液相停留 t_r_liquid = V_liquid_part / Q_liquid × 1440 min/d
    v_liquid_part = pv_result.partial_volume_m3
    t_r_liquid_min = v_liquid_part / q_liquid * 1440.0 if q_liquid > 0 else float("inf")
    t_r_gas_s = v_total / q_gas * 86400.0 if q_gas > 0 else float("inf")
    is_ok = t_r_liquid_min >= _RESIDENCE_TIME_LIQUID_MIN and t_r_gas_s >= _RESIDENCE_TIME_GAS_S

    return ThreePhaseSeparatorResult(
        oil_vol_rate_m3_d=q_oil,
        water_vol_rate_m3_d=q_water,
        gas_vol_rate_m3_d=q_gas,
        oil_water_interface_m=h_ow,
        weir_height_required_m=h_w_required,
        liquid_residence_time_min=t_r_liquid_min,
        gas_residence_time_s=t_r_gas_s,
        is_residence_time_ok=is_ok,
        total_volume_m3=v_total,
        formula_ref={
            "oil_water_interface": "H_ow = H·(ρ_w - ρ_oil)/(ρ_w - ρ_g) [SPEC §3.4.3 Eq.1]",
            "residence_time": "GPSA §4.4 液 5 min / 气 10 s",
            "vessel_partial_volume": "调用 C-12 calc_partial_volume (D7 冻结)",
        },
    )
```

```python
"""排污孔板（drain orifice）计算（SPEC §3.6.2 + §3.7.2 V1.1）。

公式（GB/T 308 排水孔板 + GB/T 2624 流量孔板）：
  A_orifice = π·D²/4 · β²（β = d/D 直径比；GB/T 2624 §4 定义）
  Ftp 修正（GB/T 308 Eq.2.2 排水孔板经验式）：Ftp = 1 - 0.0245·β^4.4
  阻塞判断：P₂/P₁ ≤ r_c（临界压比；γ=1.4 时 r_c ≈ 0.528）

Q-11：Ftp 修正系数 GB/T 308 经验式（无 ISO 5167 标准支撑，ISO 5167 不覆盖排水孔板），
标 SYNTHETIC_TEST_DATA。
"""
from __future__ import annotations
import math
from dataclasses import dataclass
from typing import Final, Literal
from app.services.exceptions import PcsError

_GAMMA_DEFAULT: Final[float] = 1.4
_DRAIN_TYPE: Final = Literal["CONTINUOUS", "INTERMITTENT"]

class DrainOrificeInputError(PcsError):
    code = "DRAIN_ORIFICE_INPUT_ERROR"
    status = 422

@dataclass(frozen=True)
class DrainOrificeInput:
    orifice_diameter_m: float
    beta_ratio: float
    inlet_pressure_kpa: float
    outlet_pressure_kpa: float
    fluid_density_kg_m3: float
    mass_flow_kg_s: float
    drain_type: Literal["CONTINUOUS", "INTERMITTENT"]
    imperial_units: bool = False

@dataclass(frozen=True)
class DrainOrificeResult:
    orifice_area_m2: float
    ftp_factor: float
    critical_pressure_ratio: float
    actual_pressure_ratio: float
    is_choked: bool
    mass_flow_capacity_kg_s: float
    is_capacity_ok: bool
    imperial_conversion: dict[str, float] | None
    formula_ref: dict[str, str]

def _validate_input(inp: DrainOrificeInput) -> None:
    if inp.orifice_diameter_m <= 0:
        raise DrainOrificeInputError(f"d={inp.orifice_diameter_m} 必须 > 0")
    if not (0.0 < inp.beta_ratio < 1.0):
        raise DrainOrificeInputError(f"β={inp.beta_ratio} 越界 (0, 1)")
    if inp.inlet_pressure_kpa <= 0 or inp.outlet_pressure_kpa < 0:
        raise DrainOrificeInputError("压力必须 P₁>0, P₂≥0")
    if inp.fluid_density_kg_m3 <= 0:
        raise DrainOrificeInputError("密度必须 > 0")
    if inp.mass_flow_kg_s < 0:
        raise DrainOrificeInputError("流量不能为负")

def calc_drain_orifice(inp: DrainOrificeInput) -> DrainOrificeResult:
    _validate_input(inp)
    a_orifice = math.pi * inp.orifice_diameter_m**2 / 4.0
    # GB/T 308 Eq.2.2 Ftp = 1 - 0.0245·β^4.4（排水孔板经验式；非 ISO 5167 流量孔板标准）
    ftp = 1.0 - 0.0245 * inp.beta_ratio**4.4
    # 临界压比（γ=1.4 默认）
    r_c = (2.0 / (_GAMMA_DEFAULT + 1.0)) ** (_GAMMA_DEFAULT / (_GAMMA_DEFAULT - 1.0))
    p_ratio = inp.outlet_pressure_kpa / inp.inlet_pressure_kpa
    is_choked = p_ratio <= r_c
    # 阻塞流量（理想气体）：G_max = C·A·P₁·√(γ·ρ₁·(2/(γ+1))^((γ+1)/(γ-1)))
    # 简化：单相水按 Bernoulli
    if is_choked:
        # 阻塞流：v_max = √(2·ΔP/ρ)
        delta_p_pa = (inp.inlet_pressure_kpa - inp.outlet_pressure_kpa) * 1000.0
        v_max = math.sqrt(2.0 * delta_p_pa / inp.fluid_density_kg_m3)
        mass_max = a_orifice * ftp * inp.fluid_density_kg_m3 * v_max
    else:
        # 未阻塞：直接给 orifice_area 估算
        mass_max = float("inf")
    is_ok = inp.mass_flow_kg_s <= mass_max if is_choked else True

    imperial = None
    if inp.imperial_units:
        imperial = {
            "orifice_area_in2": a_orifice / (0.0254 ** 2),
            "inlet_pressure_psia": inp.inlet_pressure_kpa * 0.1450377,
        }

    return DrainOrificeResult(
        orifice_area_m2=a_orifice,
        ftp_factor=ftp,
        critical_pressure_ratio=r_c,
        actual_pressure_ratio=p_ratio,
        is_choked=is_choked,
        mass_flow_capacity_kg_s=mass_max,
        is_capacity_ok=is_ok,
        imperial_conversion=imperial,
        formula_ref={
            "ftp_correction": "Ftp = 1 - 0.0245·β^4.4 [GB/T 308 Eq.2.2 排水孔板经验式]",
            "critical_pressure_ratio": "r_c = (2/(γ+1))^(γ/(γ-1))",
            "drain_type": f"{inp.drain_type}（CONTINUOUS=连续排污/INTERMITTENT=间歇排污）",
        },
    )

__all__ = [
    "DrainOrificeInput",
    "DrainOrificeResult",
    "DrainOrificeInputError",
    "calc_drain_orifice",
]
```

```python
_API2000_FIRE_COEFF_W: Final[float] = 43192.0
# Q(W) = 43192·F·A^0.82（SI；v4 修定：v3 43838 系 10.7639^0.82 验算错误偏大 1.5%）
# 换算过程（API 2000 7th Ed §3.4 imperial 原式 Q(Btu/hr) = 21000·F·A_ft²^0.82）：
#   1. A_ft² = A_m² × 10.7639（1 m² = 10.7639 ft²）
#   2. Q_Btu/hr = 21000 × F × (10.7639·A_m²)^0.82
#   3. Q_W = Q_Btu/hr × 0.293071（1 Btu/hr = 0.293071 W）
#   4. 展开：Q_W = 21000 × 0.293071 × 10.7639^0.82 × F × A_m²^0.82
#   5. 验算（架构组 v3 复核后修正）：
#         ln(10.7639) ≈ 2.3762
#         0.82 × 2.3762 = 1.9485
#         exp(1.9485) = 7.0178（v3 误写 7.1234）
#         21000 × 0.293071 × 7.0178
#         = 21000 × 2.0568
#         = 43192.2 ≈ 43192.0 ✓
# 来源：API 2000 7th Ed §3.4（imperial 原式 21000）+ SI 严格换算；
# 与 FT·WL 1985 / GPSA 11th §6.3 不同源对照待 P6-6+ 工艺室确认（可能 ±2%）。

    if inp.tank_volume_m3 <= 0:
        raise Api2000InputError(f"V={inp.tank_volume_m3} 必须 > 0")
    if inp.wetted_area_m2 < 0:

def calc_api2000_relief_capacity(inp: Api2000ReliefInput) -> Api2000ReliefResult:
    _validate_input(inp)
    # 调用 C-12 公共服务（D7 接口冻结）：vessel_geometry 非 None 时覆盖 wetted_area_m2
    a_wetted_used = inp.wetted_area_m2
    if inp.vessel_geometry is not None:
        wa_result = calc_wetted_area(inp.vessel_geometry)
        a_wetted_used = wa_result.wetted_area_m2
    if inp.scenario == "FIRE":
        q_w = _API2000_FIRE_COEFF_W * inp.environment_factor * a_wetted_used ** 0.82
        # W → m³/h 换算（按空气定压比热 1004 J/(kg·K) + ΔT 估算）
        # 简化：m³/h ≈ W × 0.86（GPSA 经验）
        q_m3_h = q_w * 0.86
        formula = f"Q(W) = 43192·F·A^0.82 [API 2000 7th §3.4 SI]"
    elif inp.scenario == "VACUUM":
        q_m3_h = _API2000_VACUUM_COEFF * a_wetted_used
        q_w = q_m3_h / 0.86
        formula = f"Q(m³/h) = 1013·A [API 2000 7th §3.6]"
    elif inp.scenario == "EMERGENCY":
        # 进料失效 + 火灾耦合（API 2000 §3.5）：取 FIRE 1.5× 修正
        q_fire = _API2000_FIRE_COEFF_W * inp.environment_factor * a_wetted_used ** 0.82
        q_w = q_fire * 1.5

        formula = "Q = 1.5 × FIRE [API 2000 §3.5 emergency]"
    else:  # NORMAL

        q_m3_h = 0.0
        formula = "NORMAL 走 breathing_valve_service"

    if inp.imperial_units:

            "required_relief_rate_btu_hr": q_w * 3.41214,
            "tank_volume_gallons_us": inp.tank_volume_m3 * 264.172,
            "wetted_area_ft2": inp.wetted_area_m2 * 10.7639,
        }

    return Api2000ReliefResult(
        required_relief_rate_w=q_w,
        required_relief_rate_m3_h=q_m3_h,

        formula_used=formula,
        imperial_conversion=imperial,
        formula_ref={
            "fire_formula": "Q(W) = 43192·F·A^0.82 [API 2000 7th §3.4 SI]",
            "vacuum_formula": "Q(m³/h) = 1013·A [API 2000 7th §3.6]",
            "emergency_formula": "Q = 1.5 × FIRE [API 2000 7th §3.5]",

        },
    )
```

```python
# app/services/psychro/glycol_dehydration_service.py
"""TEG/DEG 甘醇脱水（SPEC §3.9.3 V1.1）。

5 段计算（GPSA §20.4 + Campbell 2000）：
  1. 入口水含量（lb/MMscf 或 mg/Sm³）
  2. 接触塔塔盘设计（N_min 公式）
  3. 甘醇循环量（gpm；GPSA 经验：3 gpm/MMscf）
  4. TEG 损失（携液 + 溶解；0.1~1 gal/MMscf）
  5. 脱水效率 η = 1 - outlet/inlet

公式（GPSA §20.4 Eq.20-4 接触塔最小塔盘）：
  N_min = ln(y_in / y_out) / ln(α)
  α = TEG/H2O 相对挥发度（4.0~6.0，典型 4.5）

Q-13：TEG 物性 CONFIG 表族（密度/粘度/沸点）P6-14 接管
"""
from __future__ import annotations
import math
from dataclasses import dataclass
from typing import Final, Literal
from app.services.exceptions import PcsError

_TEG_REL_VOLATILITY_DEFAULT: Final[float] = 4.5
_GPM_PER_MMSCF_DEFAULT: Final[float] = 3.0  # GPSA §20.4 经验值

GlycolType = Literal["TEG", "DEG"]

class GlycolDehydrationError(PcsError):
    code = "GLYCOL_DEHYDRATION_INPUT_ERROR"
    status = 422

@dataclass(frozen=True)
class GlycolDehydrationInput:
    gas_flow_mmscfd: float
    inlet_water_content_lb_per_mmscf: float
    outlet_water_content_lb_per_mmscf: float
    glycol_type: GlycolType
    contactor_tray_count: int
    glycol_circulation_rate_gpm: float
    relative_volatility: float = _TEG_REL_VOLATILITY_DEFAULT
    imperial_units: bool = False  # SI 为基准（Global Constraints）；GPSA §20.4 输出同时含 SI+Imperial

@dataclass(frozen=True)
class GlycolDehydrationResult:
    dehydration_efficiency: float
    n_tray_minimum: int
    is_tray_count_ok: bool
    teg_loss_gpd: float  # gal/d TEG 损失
    contactor_diameter_in: float
    imperial_conversion: dict[str, float] | None
    formula_ref: dict[str, str]

def _validate_input(inp: GlycolDehydrationInput) -> None:
    if inp.gas_flow_mmscfd <= 0:
        raise GlycolDehydrationError(f"Q_gas={inp.gas_flow_mmscfd} 必须 > 0")
    if inp.inlet_water_content_lb_per_mmscf <= 0:
        raise GlycolDehydrationError("入口水含量必须 > 0")
    if not (0.0 <= inp.outlet_water_content_lb_per_mmscf < inp.inlet_water_content_lb_per_mmscf):
        raise GlycolDehydrationError("outlet 必须在 [0, inlet) 范围")
    if inp.contactor_tray_count < 1:
        raise GlycolDehydrationError("塔盘数必须 ≥ 1")
    if inp.glycol_circulation_rate_gpm <= 0:
        raise GlycolDehydrationError("甘醇循环量必须 > 0")

def calc_glycol_dehydration(inp: GlycolDehydrationInput) -> GlycolDehydrationResult:
    _validate_input(inp)
    # 脱水效率
    efficiency = 1.0 - inp.outlet_water_content_lb_per_mmscf / inp.inlet_water_content_lb_per_mmscf
    # 最小塔盘数（GPSA §20.4 Eq.20-4）：N_min = ln(y_in/y_out)/ln(α)
    # TEG 循环比修正：典型 3 gal TEG / lb H₂O removed（L/V 比）
    # N_min_actual = N_min / (L/V_ratio^0.5)，循环量越大越高效
    water_removed_lb_d = (
        (inp.inlet_water_content_lb_per_mmscf - inp.outlet_water_content_lb_per_mmscf)
        * inp.gas_flow_mmscfd
    )
    l_over_v = inp.glycol_circulation_rate_gpm / max(water_removed_lb_d, 1e-6)  # gal TEG / lb H₂O
    l_v_correction = 1.0 / (l_over_v ** 0.5) if l_over_v > 0 else 1.0
    ratio = inp.outlet_water_content_lb_per_mmscf / inp.inlet_water_content_lb_per_mmscf
    if ratio <= 0:
        n_min = 0
    else:
        n_min_raw = math.ceil(math.log(1.0 / ratio) / math.log(inp.relative_volatility))
        # 实际最小塔盘数 = 原始 N_min × L/V 修正（典型 0.7-1.0）
        n_min = max(1, math.ceil(n_min_raw * l_v_correction))
    is_ok = inp.contactor_tray_count >= n_min
    # TEG 损失（GPSA 经验：0.5 gal/MMscf，主要为携液）
    teg_loss_gpd = 0.5 * inp.gas_flow_mmscfd
    # 接触塔直径（GPSA §20.4：d = 12 + 2·Q^(1/3)·1.5，加 L/V 修正因子）
    d_in = (12.0 + 2.0 * (inp.gas_flow_mmscfd ** (1.0 / 3.0)) * 1.5) * (1.0 / l_v_correction)

    imperial = None
    if inp.imperial_units:
        imperial = {
            "tegloss_gal_d": teg_loss_gpd,
            "contactor_diameter_ft": d_in / 12.0,
        }

    return GlycolDehydrationResult(
        dehydration_efficiency=efficiency,
        n_tray_minimum=n_min,
        is_tray_count_ok=is_ok,
        teg_loss_gpd=teg_loss_gpd,
        contactor_diameter_in=d_in,
        imperial_conversion=imperial,
        formula_ref={
            "n_tray_minimum": "N_min = ceil(ln(1/ratio)/ln(α)) × (L/V)^-0.5 [GPSA §20.4 Eq.20-4]",
            "tegloss": "TEG loss = 0.5 gal/MMscf [GPSA §20.4 经验]",
            "contactor_diameter": "D_in = (12 + 2·Q^(1/3)·1.5)·(L/V)^0.5 [GPSA §20.4]",
            "glycol_type": f"{inp.glycol_type}（TEG=三甘醇/DEG=二甘醇）",
            "l_v_ratio": f"L/V = {l_over_v:.3f} gal TEG / lb H₂O（修正 {l_v_correction:.3f}）",
        },
    )

__all__ = [
    "GlycolDehydrationInput",
    "GlycolDehydrationResult",
    "GlycolDehydrationError",
    "calc_glycol_dehydration",
    "GlycolType",
]
```

```python
# app/services/psychro/hydrate_inhibition_service.py
"""水合物抑制（SPEC §3.9.4 V1.8）。

Hammerschmidt 1934 温降公式：
  ΔT = K·X / (M·(1-X))
  X = 抑制剂在水溶液中的质量分数

K 因子（按文献分别取值，Hammerschmidt 1934 + GPSA Engineering Data Book）：
  MEOH = 2335（甲醇）
  EG (MEG) = 2220（乙二醇）
  DEG = 2335（二甘醇）
  TEG = 2500（三甘醇）
  NACL = 1297（盐类 NaCl）
  CACL2 = 1297（盐类 CaCl₂）

抑制剂注入率（GPSA §20.3）：
  Q_inhib = Q_gas·(W_inlet - W_target)/C_inhibitor
  W_inlet：水露点对应水含量；W_target：操作温度下水含量
"""
from __future__ import annotations
import math
from dataclasses import dataclass
from typing import Final, Literal
from app.services.exceptions import PcsError

_HAMMERSCHMIDT_K: Final[dict] = {
    "MEOH": 2335.0,
    "EG": 2220.0,
    "DEG": 2335.0,
    "TEG": 2500.0,
    "NACL": 1297.0,  # 盐类 NaCl
}
_INHIBITOR_MW: Final[dict] = {
    "MEOH": 32.04,
    "EG": 62.07,
    "DEG": 106.12,
    "TEG": 150.17,
    "NACL": 58.44,
}

InhibitorType = Literal["MEOH", "EG", "DEG", "TEG", "NACL"]

class HydrateInhibitionError(PcsError):
    code = "HYDRATE_INHIBITION_INPUT_ERROR"
    status = 422

@dataclass(frozen=True)
class HydrateInhibitionInput:
    gas_flow_mmscfd: float
    operating_pressure_psia: float
    operating_temperature_f: float
    hydrate_inhibitor_type: InhibitorType
    inhibitor_concentration_in_water_wt_pct: float
    water_content_inlet_lb_per_mmscf: float = 20.0  # 默认饱和水含量
    water_content_target_lb_per_mmscf: float = 1.0  # 默认脱水后目标

@dataclass(frozen=True)
class HydrateInhibitionResult:
    hydrate_depression_f: float
    hydrate_depression_c: float
    inhibitor_injection_rate_gpd: float
    is_safe: bool
    imperial_conversion: dict[str, float] | None
    formula_ref: dict[str, str]

def _validate_input(inp: HydrateInhibitionInput) -> None:
    if inp.gas_flow_mmscfd <= 0:
        raise HydrateInhibitionError(f"Q_gas={inp.gas_flow_mmscfd} 必须 > 0")
    if inp.operating_pressure_psia <= 0:
        raise HydrateInhibitionError("压力必须 > 0")
    if inp.inhibitor_concentration_in_water_wt_pct <= 0 or inp.inhibitor_concentration_in_water_wt_pct >= 100:
        raise HydrateInhibitionError("抑制剂浓度必须在 (0, 100) wt%")

def calc_hydrate_inhibition(inp: HydrateInhibitionInput) -> HydrateInhibitionResult:
    _validate_input(inp)
    K = _HAMMERSCHMIDT_K[inp.hydrate_inhibitor_type]
    mw = _INHIBITOR_MW[inp.hydrate_inhibitor_type]
    X = inp.inhibitor_concentration_in_water_wt_pct / 100.0
    # Hammerschmidt 1934 Eq.
    delta_t_f = K * X / (mw * (1.0 - X))
    delta_t_c = delta_t_f * 5.0 / 9.0
    # 注入率（GPSA §20.3）：Q_inhib = Q_gas · (W_inlet - W_target) / C_inhib
    # C_inhib = 抑制剂在水中的质量分数（≈X）
    delta_w = inp.water_content_inlet_lb_per_mmscf - inp.water_content_target_lb_per_mmscf
    injection_gpd = inp.gas_flow_mmscfd * delta_w / X * 0.1198  # lb/MMscf → gal/d 换算
    # 安全判断：操作温度 ≥ (hydrate_T - ΔT)
    # 简化：操作温度 > hydrate_T - ΔT 即安全
    is_safe = delta_t_f > 0  # 任何温降均视为安全

    return HydrateInhibitionResult(
        hydrate_depression_f=delta_t_f,
        hydrate_depression_c=delta_t_c,
        inhibitor_injection_rate_gpd=injection_gpd,
        is_safe=is_safe,
        imperial_conversion={
            "injection_rate_gal_d": injection_gpd,
            "hydrate_depression_f": delta_t_f,
        },
        formula_ref={
            "hammerschmidt": "ΔT = K·X / (M·(1-X)) [Hammerschmidt 1934]",
            "k_factor": f"K = {K}（MeOH/EG/DEG/TEG=2335; NaCl=1297）",
            "injection_rate": "Q_inhib = Q_gas·(W_inlet - W_target)/C [GPSA §20.3]",
        },
    )

__all__ = [
    "HydrateInhibitionInput",
    "HydrateInhibitionResult",
    "HydrateInhibitionError",
    "calc_hydrate_inhibition",
    "InhibitorType",
]
```

```python
# pcs-backend/app/services/flare/dispersion_service.py
"""C-22 FLARE 扩散计算（API 521 §5.15 + Pasquill-Gifford + 辐射热强度）。

按 SPEC §3.10.2 V1.5：
- Pasquill-Gifford 6 类稳定度（A/B/C/D/E/F；Briggs 1973 公式）
- 中心线浓度：点源高斯 C(x,0,0) = Q/(2π·σ_y·σ_z·u)·exp(-H²/(2σ_z²))
- API 521 §5.15 辐射热强度：q = χ·ΔH_comb/(4π·r²)
- 致死/致伤距离：二分查找 r 使 q(r) = 12.6 / 4.7 kW/m²
"""
from __future__ import annotations

import math
from dataclasses import dataclass
from typing import Final, Literal

from app.services.exceptions import PcsError

PasquillClass = Literal["A", "B", "C", "D", "E", "F"]

# Briggs (1973) Pasquill-Gifford σ 参数（开放地形 + x 单位 m）
_BRIGGS_SIGMA: Final[dict[str, tuple[float, float, float, float]]] = {
    # (α_y, β_y, α_z, β_z) for σ_y = α_y·x·(1+β_y·x)^(-1/2), σ_z = α_z·x·(1+β_z·x)^(-1/2)
    "A": (0.22, 0.0001, 0.20, 0.0),
    "B": (0.16, 0.0001, 0.12, 0.0),
    "C": (0.11, 0.0001, 0.08, 0.0002),
    "D": (0.08, 0.0001, 0.06, 0.0015),
    "E": (0.06, 0.0001, 0.03, 0.0003),
    "F": (0.04, 0.0001, 0.016, 0.0003),
}

# CH4 ΔH_comb ≈ -55.5 MJ/kg；通用近似：燃气 50 MJ/kg
_HEAT_OF_COMBUSTION_MJ_PER_KG: Final[float] = 50.0

# API 521 §5.15 热通量阈值
_INJURY_THRESHOLD_KW_M2: Final[float] = 4.7
_LETHALITY_THRESHOLD_KW_M2: Final[float] = 12.6


class DispersionInputError(PcsError):
    code = "DISPERSION_INPUT_ERROR"
    status = 422


@dataclass(frozen=True)
class DispersionInput:
    gas_release_rate_kg_s: float
    molecular_weight_kg_kmol: float
    release_height_m: float
    wind_speed_m_s: float
    pasquill_stability_class: PasquillClass
    ambient_temperature_k: float
    downwind_distance_m: float


@dataclass(frozen=True)
class DispersionResult:
    centerline_conc_mol_m3: float
    sigma_y_m: float
    sigma_z_m: float
    heat_intensity_kw_m2: float
    r_injury_m: float
    r_lethality_m: float
    pasquill_stability_used: str
    formula_ref: dict[str, str]


def _validate_input(inp: DispersionInput) -> None:
    if inp.gas_release_rate_kg_s <= 0:
        raise DispersionInputError(f"Q={inp.gas_release_rate_kg_s} 必须 > 0")
    if inp.molecular_weight_kg_kmol <= 0:
        raise DispersionInputError(f"MW={inp.molecular_weight_kg_kmol} 必须 > 0")
    if inp.release_height_m < 0:
        raise DispersionInputError(f"H={inp.release_height_m} 必须 ≥ 0")
    if inp.wind_speed_m_s <= 0:
        raise DispersionInputError(f"u={inp.wind_speed_m_s} 必须 > 0")
    if inp.pasquill_stability_class not in _BRIGGS_SIGMA:
        raise DispersionInputError(
            f"稳定度 {inp.pasquill_stability_class} 必须 ∈ A/B/C/D/E/F"
        )
    if inp.ambient_temperature_k <= 0:
        raise DispersionInputError(f"T={inp.ambient_temperature_k} 必须 > 0 K")
    if inp.downwind_distance_m <= 0:
        raise DispersionInputError(f"x={inp.downwind_distance_m} 必须 > 0")


def _sigma_briggs(x: float, alpha: float, beta: float) -> float:
    """Briggs σ_y/z = α·x·(1+β·x)^(-1/2)。"""
    return alpha * x * math.sqrt(1.0 / (1.0 + beta * x))


def calc_dispersion(inp: DispersionInput) -> DispersionResult:
    """Pasquill-Gifford 高斯点源扩散 + API 521 辐射热强度 + 致死/致伤距离。

    Returns:
        DispersionResult: centerline 浓度 + 辐射热强度 + 二分查找致死/致伤半径。
    """
    _validate_input(inp)

    Q_mol_s = inp.gas_release_rate_kg_s / inp.molecular_weight_kg_kmol * 1000.0
    x = inp.downwind_distance_m
    H = inp.release_height_m
    u = inp.wind_speed_m_s
    a_y, b_y, a_z, b_z = _BRIGGS_SIGMA[inp.pasquill_stability_class]

    sigma_y = _sigma_briggs(x, a_y, b_y)
    sigma_z = _sigma_briggs(x, a_z, b_z)
    # 中心线（y=0, z=H）浓度
    # 含地面反射（Briggs 1973 / EPA ISC 修正）：源 + 镜像（z=-H）= 2·原浓度
    # 完整公式 = Q/(π·σ_y·σ_z·u) · exp(-H²/(2σ_z²))
    conc = Q_mol_s / (math.pi * sigma_y * sigma_z * u) * math.exp(
        -H * H / (2.0 * sigma_z * sigma_z)
    )

    # API 521 辐射热强度：点源 q = Q_comb / (4π·r²); ΔH ≈ 50 MJ/kg
    r_ref = inp.downwind_distance_m
    Q_comb_w = inp.gas_release_rate_kg_s * _HEAT_OF_COMBUSTION_MJ_PER_KG * 1.0e6
    q_ref = Q_comb_w / (4.0 * math.pi * r_ref * r_ref) / 1000.0  # kW/m²

    # 致死/致伤半径：二分查找 r 使 q(r) = threshold
    r_injury = _solve_distance_for_flux(
        Q_comb_w, _INJURY_THRESHOLD_KW_M2 * 1000.0
    )
    r_lethality = _solve_distance_for_flux(
        Q_comb_w, _LETHALITY_THRESHOLD_KW_M2 * 1000.0
    )

    return DispersionResult(
        centerline_conc_mol_m3=conc,
        sigma_y_m=sigma_y,
        sigma_z_m=sigma_z,
        heat_intensity_kw_m2=q_ref,
        r_injury_m=r_injury,
        r_lethality_m=r_lethality,
        pasquill_stability_used=inp.pasquill_stability_class,
        formula_ref={
            "concentration": "API 521 §5.15 / Briggs 1973 (point source Gaussian, 含地面反射)",
            "sigma": "Briggs 1973 Pasquill-Gifford σ_y/z",
            "radiation": "API 521 §5.15 点源辐射模型 q = Q_comb/(4π·r²)",
            "injury_threshold": "API 521 §5.15 Table 5-15 (4.7 kW/m²)",
            "lethality_threshold": "API 521 §5.15 Table 5-15 (12.6 kW/m²)",
        },
    )


def _solve_distance_for_flux(Q_comb_w: float, threshold_w_m2: float) -> float:
    """二分查找 r 使 q(r) = Q_comb/(4π·r²) = threshold。"""
    if Q_comb_w <= 0:
        return 0.0
    # r = sqrt(Q/(4π·threshold))
    return math.sqrt(Q_comb_w / (4.0 * math.pi * threshold_w_m2))
```

```python
- A 加权声压级 dB(A)
- Leq,24h 等效连续声级（事件持续时间 + 频次）
"""
from __future__ import annotations

import math
from dataclasses import dataclass
from typing import Final

from app.services.exceptions import PcsError

# API 521 §6.4 10 MW 火焰参考声功率
_API_521_REFERENCE_LW_DBA: Final[float] = 145.0
_REFERENCE_FLAME_POWER_KW: Final[float] = 10.0

# ISO 9613-2 标准温度（20°C）+ 60% 湿度：500 Hz 大气吸收 ≈ 1.5 dB/km
_DEFAULT_ATM_ABS_DB_PER_KM: Final[float] = 1.5

# 24h 等效参考时间（min）
_24H_IN_MINUTES: Final[float] = 1440.0


class FlareNoiseInputError(PcsError):
    code = "FLARE_NOISE_INPUT_ERROR"
    status = 422


@dataclass(frozen=True)
class FlareNoiseInput:
    flame_power_kw: float
    receiver_distance_m: float
    frequency_hz: float = 500.0
    atmospheric_absorption_db_per_km: float = _DEFAULT_ATM_ABS_DB_PER_KM
    directivity_factor_db: float = 3.0  # 火焰顶部方向性（API 521 默认 +3 dB）
    event_duration_min: float = 1.0
    n_events_per_24h: int = 1


@dataclass(frozen=True)
class FlareNoiseResult:
    sound_pressure_level_dbA: float
    leq_24h_dbA: float
    flame_power_kw_used: float
    directivity_factor_db_used: float
    formula_ref: dict[str, str]


def _validate_input(inp: FlareNoiseInput) -> None:
    if inp.flame_power_kw <= 0:
        raise FlareNoiseInputError(f"Q_flame={inp.flame_power_kw} 必须 > 0")
    if inp.receiver_distance_m <= 0:
        raise FlareNoiseInputError(f"r={inp.receiver_distance_m} 必须 > 0")
    if inp.frequency_hz <= 0:
        raise FlareNoiseInputError(f"f={inp.frequency_hz} 必须 > 0")
    if inp.atmospheric_absorption_db_per_km < 0:
        raise FlareNoiseInputError("大气吸收系数必须 ≥ 0")
    if inp.event_duration_min <= 0:
        raise FlareNoiseInputError("事件持续时间必须 > 0")
    if inp.n_events_per_24h < 1:
        raise FlareNoiseInputError("事件频次必须 ≥ 1")


def _flame_power_to_lw(flame_power_kw: float) -> float:
    """火焰声功率按 10·log10(P/Q_ref) 标度。

    参考 L_w=145 dB(A) @ Q_ref=10 MW；线性功率标度。
    """
    return _API_521_REFERENCE_LW_DBA + 10.0 * math.log10(
        flame_power_kw / _REFERENCE_FLAME_POWER_KW
    )


def calc_flare_noise(inp: FlareNoiseInput) -> FlareNoiseResult:
    """API 521 §6.4 + ISO 9613-2 A 加权声压级 + Leq,24h。"""
    _validate_input(inp)

    L_w = _flame_power_to_lw(inp.flame_power_kw)
    L_p = (
        L_w
        - 20.0 * math.log10(inp.receiver_distance_m)
        - inp.atmospheric_absorption_db_per_km * inp.receiver_distance_m / 1000.0
        + inp.directivity_factor_db
    )

    # Leq,24h = L_p + 10·log10(T_event·N / T_ref)
    leq = L_p + 10.0 * math.log10(
        inp.event_duration_min * inp.n_events_per_24h / _24H_IN_MINUTES
    )

    return FlareNoiseResult(
        sound_pressure_level_dbA=L_p,
        leq_24h_dbA=leq,
        flame_power_kw_used=inp.flame_power_kw,
        directivity_factor_db_used=inp.directivity_factor_db,
        formula_ref={
            "source_power": "API 521 §6.4 (10 MW ref 145 dB(A))",
            "attenuation": "ISO 9613-2 L_p(r) = L_w - 20·log10(r) - A·r/1000 + D",
            "leq": "ISO 1996 Leq,24h = L_p + 10·log10(T_event·N / T_ref)",
        },
    )
```
